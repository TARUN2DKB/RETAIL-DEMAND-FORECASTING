def calculate_metrics(
    y_true,
    y_pred
):

    y_true = np.asarray(
        y_true,
        dtype=np.float64
    )

    y_pred = np.asarray(
        y_pred,
        dtype=np.float64
    )

    # Demand cannot be negative
    y_pred = np.clip(
        y_pred,
        0,
        None
    )

    # -------------------------
    # MAE
    # -------------------------
    mae = np.mean(
        np.abs(
            y_true - y_pred
        )
    )

    # -------------------------
    # RMSE
    # -------------------------
    rmse = np.sqrt(
        np.mean(
            (
                y_true -
                y_pred
            ) ** 2
        )
    )

    # -------------------------
    # WAPE
    # -------------------------
    denominator = np.sum(
        np.abs(y_true)
    )

    if denominator > 0:

        wape = (
            np.sum(
                np.abs(
                    y_true -
                    y_pred
                )
            )
            /
            denominator
        ) * 100

    else:

        wape = np.nan

    # -------------------------
    # sMAPE
    # -------------------------
    smape_denominator = (
        np.abs(y_true)
        +
        np.abs(y_pred)
    )

    smape_numerator = (
        2 *
        np.abs(
            y_true -
            y_pred
        )
    )

    smape_array = np.divide(
        smape_numerator,
        smape_denominator,
        out=np.zeros_like(
            smape_numerator,
            dtype=np.float64
        ),
        where=(
            smape_denominator != 0
        )
    )

    smape = (
        np.mean(
            smape_array
        ) * 100
    )

    return {
        "MAE": float(mae),
        "RMSE": float(rmse),
        "WAPE": float(wape),
        "sMAPE": float(smape)
    }

    model_name,

def create_seasonal_naive(
    X
):

    n = X.shape[0]

    predictions = np.zeros(
        (
            n,
            HORIZON
        ),
        dtype=np.float32
    )

    lag7 = X[:, lag7_idx]
    lag14 = X[:, lag14_idx]
    lag28 = X[:, lag28_idx]

    # Forecast days 1-7
    predictions[:, 0:7] = (
        lag7[:, None]
    )

    # Forecast days 8-14
    predictions[:, 7:14] = (
        lag14[:, None]
    )

    # Forecast days 15-28
    predictions[:, 14:28] = (
        lag28[:, None]
    )

    predictions = np.clip(
        predictions,
        0,
        None
    )

    return predictions

def create_prediction_dataframe(
    source_df,
    y_true,
    y_pred,
    model_name
):

    records = []

    # Prefer series_key
    if "series_key" in source_df.columns:

        series_ids = (
            source_df[
                "series_key"
            ].astype(str).values
        )

    else:

        series_ids = (
            source_df[
                "id"
            ].astype(str).values
        )

    for i in range(
        len(series_ids)
    ):

        for h in range(
            HORIZON
        ):

            records.append({
                "series_id":
                    series_ids[i],

                "forecast_day":
                    h + 1,

                "actual":
                    float(
                        y_true[i, h]
                    ),

                "prediction":
                    float(
                        y_pred[i, h]
                    ),

                "model":
                    model_name
            })

    return pd.DataFrame(
        records
    )

    y_true,
    y_pred
):

    return np.mean(
        np.abs(
            y_true -
            y_pred
        ),
        axis=0
    )
